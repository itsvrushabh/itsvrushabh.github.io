14     pub fn new(initial: S) -> (Self, broadcast::Receiver<E>) {
15         let (tx, rx) = broadcast::channel(1024);

17     }

19     pub async fn transition(&self, event: E, next: S) -> Result<(), &'static str> {
20         let mut lock = self.state.write().await;
21         if *lock == next { return Err("LoopDetected"); }
22         self.history.write().await.push((lock.clone(), event.clone()));
23         *lock = next;
24         let _ = self.tx.send(event); // Zero-allocation broadcast
25         Ok(())
